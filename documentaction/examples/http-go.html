HTTP Examples in Go

Copy-paste examples for using TsunamiDB HTTP API from Go with the standard net/http package. Subscription WebSocket examples use github.com/gorilla/websocket.

Basic Key-Value

const baseURL = "http://localhost:5844"

func Save(table, key string, data []byte) error {
	url := fmt.Sprintf("%s/save/%s/%s", baseURL, table, key)
	resp, err := http.Post(url, "application/octet-stream", bytes.NewReader(data))
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		body, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("save failed: %s: %s", resp.Status, string(body))
	}
	return nil
}

func Read(table, key string) ([]byte, error) {
	url := fmt.Sprintf("%s/read/%s/%s", baseURL, table, key)
	resp, err := http.Get(url)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		body, _ := io.ReadAll(resp.Body)
		return nil, fmt.Errorf("read failed: %s: %s", resp.Status, string(body))
	}
	return io.ReadAll(resp.Body)
}

type PatchOp struct {
	Offset    int    `json:"offset"`
	Delete    int    `json:"delete,omitempty"`
	Insert    string `json:"insert,omitempty"`
	InsertB64 string `json:"insert_b64,omitempty"`
}

func Patch(table, key string, ops []PatchOp) ([]byte, error) {
	url := fmt.Sprintf("%s/patch/%s/%s", baseURL, table, key)
	payload, err := json.Marshal(map[string][]PatchOp{"ops": ops})
	if err != nil {
		return nil, err
	}

	resp, err := http.Post(url, "application/json", bytes.NewReader(payload))
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	body, _ := io.ReadAll(resp.Body)
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("patch failed: %s: %s", resp.Status, string(body))
	}
	return body, nil
}

Nested JSON

func SaveNested(table, key string, body []byte) error {
	url := fmt.Sprintf("%s/save/%s/%s", baseURL, table, key)
	req, err := http.NewRequest(http.MethodPost, url, bytes.NewReader(body))
	if err != nil {
		return err
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("mode", "save_nested_json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		errBody, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("save nested failed: %s: %s", resp.Status, string(errBody))
	}
	return nil
}

func ReadNested(table, key string, paths []string, only bool) ([]byte, error) {
	url := fmt.Sprintf("%s/read/%s/%s", baseURL, table, key)
	req, err := http.NewRequest(http.MethodGet, url, nil)
	if err != nil {
		return nil, err
	}

	if len(paths) > 0 {
		header := "read_nested"
		if only {
			header = "read_only_nested"
		}
		req.Header.Set(header, strings.Join(paths, " "))
	}

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		body, _ := io.ReadAll(resp.Body)
		return nil, fmt.Errorf("read nested failed: %s: %s", resp.Status, string(body))
	}
	return io.ReadAll(resp.Body)
}

Incremental Tables

type SaveIncResponse struct {
	ID      string `json:"id"`
	Warning string `json:"warning,omitempty"`
}

type IncEntry struct {
	ID   uint64 `json:"id"`
	Data string `json:"data"`
}

func SaveIncAppend(table, key string, maxEntrySize uint64, entryKey string, data []byte) (*SaveIncResponse, error) {
	url := fmt.Sprintf("%s/save_inc/%s/%s", baseURL, table, key)
	req, err := http.NewRequest(http.MethodPost, url, bytes.NewReader(data))
	if err != nil {
		return nil, err
	}
	req.Header.Set("Content-Type", "application/octet-stream")
	req.Header.Set("max_entry_size", fmt.Sprintf("%d", maxEntrySize))
	if entryKey != "" {
		req.Header.Set("entry_key", entryKey)
	}

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		body, _ := io.ReadAll(resp.Body)
		return nil, fmt.Errorf("save_inc failed: %s: %s", resp.Status, string(body))
	}

	var out SaveIncResponse
	if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
		return nil, err
	}
	return &out, nil
}

type enableSubscriptionRequest struct {
	Keys []string `json:"keys"`
}

type enableSubscriptionResponse struct {
	AuthKey string `json:"auth_key"`
}

func MintSubscriptionToken(keys []string) (string, error) {
	payload, _ := json.Marshal(enableSubscriptionRequest{Keys: keys})
	resp, err := http.Post(
		baseURL+"/subscriptions/enable",
		"application/json",
		bytes.NewReader(payload),
	)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		body, _ := io.ReadAll(resp.Body)
		return "", fmt.Errorf("enable subscription failed: %s: %s", resp.Status, string(body))
	}

	var out enableSubscriptionResponse
	if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
		return "", err
	}
	return out.AuthKey, nil
}

WebSocket Client

func ConnectSubscriptionSocket(authKey string) (*websocket.Conn, error) {
	dialer := websocket.Dialer{HandshakeTimeout: 10 * time.Second}
	conn, _, err := dialer.Dial("ws://localhost:5845/sub", nil)
	if err != nil {
		return nil, err
	}

	payload, _ := json.Marshal(map[string]string{"auth_key": authKey})
	if err := conn.WriteMessage(websocket.TextMessage, payload); err != nil {
		_ = conn.Close()
		return nil, err
	}
	return conn, nil
}

func ReadSubscriptionEvents(conn *websocket.Conn) error {
	for {
		_, msg, err := conn.ReadMessage()
		if err != nil {
			return err
		}
		fmt.Println(string(msg))
	}
}
